/** * MANGALAM — LEADERSHIP REVIEW 2026 (Backend) v4 — leaders as receivers + downward team reviews * --------------------------------------------------------------------- * Direction: the 32 leaders (+SSS/RP) are RECEIVERS. Everyone mapped is a giver. * VENDOR : giver reviews a cross-department leader * PEER : giver reviews a peer leader * TEAM : giver reviews their own reporting manager / leader (upward) * MANAGER: leader reviews their own team member (downward, with tenure question) * * Sheet tabs required: * Givers : Giver Name | Employee ID | Email | Department (32 leaders, login master) * Assignments : Giver | Receiver Key | Receiver Name | Receiver Department | Form Type | Status | Submitted At * Responses : auto-created on first submission * Config : B1 = admin pin (default 2026 if missing) * * Deploy: Web app · Execute as Me · Access: Anyone. function doGet(e) { var p = (e && e.parameter) || {}; var out; try { switch (p.action) { case 'login': out = apiLogin(p.email || ''); break; case 'submit': out = apiSubmit(p.payload || ''); break; case 'dashboard': out = apiDashboard(p.pin || ''); break; case 'audit': out = apiAudit(p.pin || ''); break; default: out = { ok: false, error: 'Unknown action' }; } } catch (err) { out = { ok: false, error: String(err) }; } var json = JSON.stringify(out); if (p.callback) { return ContentService.createTextOutput(p.callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); } /* ---------- helpers ---------- */ function normKey(s) { return String(s || '').replace(/\s+/g, ' ').trim().toLowerCase(); } function getAssignmentsData() { var sh = SS().getSheetByName('Assignments'); if (!sh) throw new Error('Assignments tab missing'); var vals = sh.getDataRange().getValues(); var rows = []; for (var i = 1; i < vals.length; i++) { var r = vals[i]; if (!r[0]) continue; rows.push({ row: i + 1, giver: String(r[0]).trim(), rkey: String(r[1]).trim(), receiver: String(r[2]).trim(), dept: String(r[3]).trim(), form: String(r[4]).trim().toUpperCase(), status: String(r[5] || 'PENDING').trim().toUpperCase() }); } return { sheet: sh, rows: rows }; } function getGiversMaster() { var sh = SS().getSheetByName('Givers'); if (!sh) throw new Error('Givers tab missing'); var vals = sh.getDataRange().getValues(); var byEmail = {}; for (var i = 1; i < vals.length; i++) { var r = vals[i]; if (!r[0]) continue; var email = String(r[2] || '').trim().toLowerCase(); if (!email || email.indexOf('@') === -1) continue; byEmail[email] = { name: String(r[0]).trim(), empId: String(r[1] || '').trim(), email: email, dept: String(r[3] || '').trim() }; } return byEmail; } function resolveEmail(email) { var e = String(email || '').trim().toLowerCase(); if (!e) return null; return getGiversMaster()[e] || null; } function getAdminPin() { var sh = SS().getSheetByName('Config'); if (sh) { var v = String(sh.getRange('B1').getValue() || '').trim(); if (v) return v; } return '2026'; } /* ---------- endpoints ---------- */ function apiLogin(email) { var g = resolveEmail(email); if (!g) return { ok: false, error: 'Email not found. Please use your registered email ID or contact HR.' }; var rows = getAssignmentsData().rows.filter(function (a) { return normKey(a.giver) === normKey(g.name); }); return { ok: true, giver: { name: g.name, empId: g.empId, dept: g.dept, email: g.email }, assignments: rows.map(function (a) { return { rkey: a.rkey, receiver: a.receiver, dept: a.dept, form: a.form, status: a.status }; }) }; } /** * payload JSON: { email, rkey, form, ratings:[..], tenure?, comments? } */ function apiSubmit(payloadStr) { var p; try { p = JSON.parse(payloadStr); } catch (e) { return { ok: false, error: 'Bad payload' }; } if (!p.email || !p.rkey || !p.form || !Array.isArray(p.ratings)) { return { ok: false, error: 'Missing fields' }; } var g = resolveEmail(p.email); if (!g) return { ok: false, error: 'Email not recognized' }; var form = String(p.form).toUpperCase(); var qList = FORM_QUESTIONS[form]; if (!qList) return { ok: false, error: 'Unknown form type' }; if (p.ratings.length !== qList.length) return { ok: false, error: 'Expected ' + qList.length + ' ratings' }; for (var i = 0; i < p.ratings.length; i++) { var v = Number(p.ratings[i]); if (!(v >= 1 && v <= 10)) return { ok: false, error: 'Ratings must be 1–10' }; } if (form === 'MANAGER' && !p.tenure) return { ok: false, error: 'Tenure selection required' }; var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var data = getAssignmentsData(); var match = null; for (var j = 0; j < data.rows.length; j++) { var a = data.rows[j]; if (normKey(a.giver) === normKey(g.name) && a.rkey === String(p.rkey).trim() && a.form === form) { match = a; break; } } if (!match) return { ok: false, error: 'Assignment not found' }; if (match.status === 'DONE') return { ok: false, error: 'Already submitted' }; var ss = SS(); var resp = ss.getSheetByName('Responses'); if (!resp) { resp = ss.insertSheet('Responses'); resp.appendRow(['Timestamp', 'Giver', 'Receiver Key', 'Receiver', 'Receiver Department', 'Form', 'Q1', 'Q2', 'Q3', 'Q4', 'Q5', 'Q6', 'Q7', 'Avg', 'Tenure', 'Comments']); resp.setFrozenRows(1); } var ratings = p.ratings.map(Number); var padded = ratings.concat([,,,,,,,]).slice(0, 7); var avg = ratings.reduce(function (s, x) { return s + x; }, 0) / ratings.length; resp.appendRow([new Date(), g.name, match.rkey, match.receiver, match.dept, form, padded[0]||'', padded[1]||'', padded[2]||'', padded[3]||'', padded[4]||'', padded[5]||'', padded[6]||'', Math.round(avg * 100) / 100, p.tenure || '', String(p.comments || '').slice(0, 2000)]); data.sheet.getRange(match.row, 6).setValue('DONE'); data.sheet.getRange(match.row, 7).setValue(new Date()); return { ok: true }; } finally { lock.releaseLock(); } } /** Full drill-down: every assignment + every individual response (admin only) */ function apiAudit(pin) { if (String(pin) !== getAdminPin()) return { ok: false, error: 'Invalid PIN' }; var assign = getAssignmentsData().rows.map(function (a) { return { giver: a.giver, rkey: a.rkey, receiver: a.receiver, dept: a.dept, form: a.form, status: a.status }; }); var responses = []; var sh = SS().getSheetByName('Responses'); if (sh && sh.getLastRow() > 1) { var vals = sh.getRange(2, 1, sh.getLastRow() - 1, 16).getValues(); vals.forEach(function (r) { var ratings = []; for (var q = 0; q < 7; q++) { var v = Number(r[6 + q]); if (v >= 1) ratings.push(v); } responses.push({ ts: r[0] ? Utilities.formatDate(new Date(r[0]), Session.getScriptTimeZone(), 'dd MMM HH:mm') : '', giver: String(r[1]), rkey: String(r[2]).trim(), receiver: String(r[3]), form: String(r[5]).trim().toUpperCase(), ratings: ratings, avg: Number(r[13]) || null, tenure: String(r[14] || ''), comments: String(r[15] || '') }); }); } return { ok: true, assignments: assign, responses: responses }; } function apiDashboard(pin) { if (String(pin) !== getAdminPin()) return { ok: false, error: 'Invalid PIN' }; var ss = SS(); var assign = getAssignmentsData().rows; var recv = {}; // by receiver key assign.forEach(function (a) { if (!recv[a.rkey]) recv[a.rkey] = { key: a.rkey, name: a.receiver, dept: a.dept, forms: { VENDOR: {t:0,d:0}, PEER: {t:0,d:0}, TEAM: {t:0,d:0}, MANAGER: {t:0,d:0} } }; var f = recv[a.rkey].forms[a.form]; if (f) { f.t++; if (a.status === 'DONE') f.d++; } }); var sh = ss.getSheetByName('Responses'); var comments = {}; if (sh && sh.getLastRow() > 1) { var vals = sh.getRange(2, 1, sh.getLastRow() - 1, 16).getValues(); vals.forEach(function (r) { var key = String(r[2]).trim(), form = String(r[5]).trim().toUpperCase(); var R = recv[key]; if (!R) return; if (!R.scores) R.scores = {}; if (!R.scores[form]) R.scores[form] = { n: 0, qsum: [0,0,0,0,0,0,0], sum: 0, tenure: {} }; var S = R.scores[form]; S.n++; var cnt = 0, tot = 0; for (var q = 0; q < 7; q++) { var v = Number(r[6 + q]); if (v >= 1) { S.qsum[q] += v; cnt++; tot += v; } } S.sum += cnt ? tot / cnt : 0; if (r[14]) S.tenure[r[14]] = (S.tenure[r[14]] || 0) + 1; if (r[15]) { if (!comments[key]) comments[key] = []; comments[key].push({ form: form, giver: String(r[1]), text: String(r[15]) }); } }); } var out = Object.keys(recv).map(function (key) { var R = recv[key]; var formAvgs = {}, wSum = 0, wN = 0; ['VENDOR','PEER','TEAM','MANAGER'].forEach(function (f) { var S = R.scores && R.scores[f]; if (S && S.n) { var avg = S.sum / S.n; formAvgs[f] = { avg: Math.round(avg * 100) / 100, n: S.n, q: S.qsum.map(function (x) { return S.n ? Math.round(x / S.n * 100) / 100 : 0; }), tenure: S.tenure }; wSum += avg * S.n; wN += S.n; } }); return { key: key, name: R.name, dept: R.dept, completion: R.forms, formAvgs: formAvgs, overall: wN ? Math.round(wSum / wN * 100) / 100 : null, responses: wN, comments: comments[key] || [] }; }); out.sort(function (a, b) { return (b.overall || 0) - (a.overall || 0); }); return { ok: true, receivers: out }; }
